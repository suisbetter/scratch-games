//----- Green flag events ---------------------------------------------------------------------------------------------

WhenGreenFlagClicked()
{
    go inside = 0;
    Looks.Hide();
}


//----- Broadcast received events -------------------------------------------------------------------------------------

WhenBroadcastReceived(door opening)
{
    Looks.Show();
    Repeat Until (go inside == 1)
    {
        Motion.GoTo(_mouse_);
    }
}

WhenBroadcastReceived(door opening)
{
    Forever
    {
        If (Sensing.MouseDown() And Sensing.TouchingObject(_mouse_))
        {
            Event.Broadcast("go inside");
            Stop(this script);
        }
    }
}

WhenBroadcastReceived(entering the funtimes)
{
    Looks.Hide();
}

WhenBroadcastReceived(primary control module)
{
    Looks.Hide();
}
