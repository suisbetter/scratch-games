//----- Green flag events ---------------------------------------------------------------------------------------------

WhenGreenFlagClicked()
{
    Motion.GoToXY(0, 0);
    Looks.SetEffectTo(GHOST, 100);
    circus control = 0;
    Looks.ClearGraphicEffects();
    Looks.SetSizeTo(150);
    Looks.Hide();
}

{
    Repeat (15)
    {
        Looks.NextCostume();
    }
}

    Wait Until (actual funtime auditorium == 0);
    Wait Until (circus control == 0);

    {
        If (Not (Sensing.KeyPressed(w)))
        {
            Wait Until (go == 1);
            Wait Until (circus control == 1);
            Sound.StopAllSounds();
        }
    }
}
