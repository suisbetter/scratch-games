//----- Green flag events ---------------------------------------------------------------------------------------------

WhenGreenFlagClicked()
{
    Looks.SwitchCostumeTo(IMG_0088);
    Looks.Hide();
}

WhenGreenFlagClicked()
{
    Forever
    {
        If ((Not ((aggression factor == 4))) Or (Not ((aggression factor == 10))))
        {
            Looks.Hide();
            Stop(this script);
        }
    }
}


//----- Key pressed events --------------------------------------------------------------------------------------------

WhenKeyPressed(space)
{
    Control.Wait(0.5);
    Wait Until (funtime auditorium == 1);
    Forever
    {
        If ((aggression factor == 4) Or (aggression factor == 10))
        {
            Looks.Show();
            Repeat (43)
            {
                Looks.NextCostume();
            }
            Looks.Hide();
            Stop(this script);
        }
    }
}


//----- Broadcast received events -------------------------------------------------------------------------------------

WhenBroadcastReceived(scooping room)
{
    Looks.Hide();
}
