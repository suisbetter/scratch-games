//----- Green flag events ---------------------------------------------------------------------------------------------

WhenGreenFlagClicked()
{
    funtime auditorium = 0;
    Looks.Hide();
}


//----- Broadcast received events -------------------------------------------------------------------------------------

WhenBroadcastReceived(circus control)
{
    Looks.Hide();
}

WhenBroadcastReceived(enter funtime auditorium)
{
    Repeat (10)
    {
        Looks.ChangeEffectBy(GHOST, 10);
    }
}

WhenBroadcastReceived(primary control module)
{
    Sound.PlayUntilDone(Door-Me_58G_HD2-29632);
    Motion.GoToXY(0, 0);
    Looks.SetSizeTo(150);
    Looks.Show();
}

WhenBroadcastReceived(primary control module)
{
    prm += 1;
    Repeat Until (funtime auditorium == 1)
    {
        Repeat (16)
        {
            Looks.NextCostume();
        }
    }
}

WhenBroadcastReceived(primary control module)
{
    Forever
    {
        Motion.SetY(((Motion.YPosition() - ((Sensing.MouseY() + Motion.YPosition()) / 40)) / 1.18));
    }
}

WhenBroadcastReceived(primary control module)
{
    Forever
    {
        Motion.SetX(((Motion.XPosition() - ((Sensing.MouseX() + Motion.XPosition()) / 40)) / 1.18));
    }
}

WhenBroadcastReceived(primary control module)
{
    Sound.PlayUntilDone(Dark_amb_v4);
}
