//----- Green flag events ---------------------------------------------------------------------------------------------

WhenGreenFlagClicked()
{
    Looks.SwitchCostumeTo(IMG_0089);
    Looks.Hide();
}


//----- Key pressed events --------------------------------------------------------------------------------------------

WhenKeyPressed(space)
{
    Forever
    {
        If ((Not ((aggression factor == 2))) Or (Not ((aggression factor == 8))))
        {
            Looks.Hide();
            Stop(this script);
        }
    }
}

WhenKeyPressed(space)
{
    Control.Wait(0.5);
    Wait Until (funtime auditorium == 1);
    Forever
    {
        If ((aggression factor == 2) Or (aggression factor == 8))
        {
            Looks.Show();
            Repeat (43)
            {
                Looks.NextCostume();
            }
            Looks.Hide();
            Stop(this script);
        }
    }
}


//----- Broadcast received events -------------------------------------------------------------------------------------

WhenBroadcastReceived(scooping room)
{
    Looks.Hide();
}
