//----- Green flag events ---------------------------------------------------------------------------------------------

WhenGreenFlagClicked()
{
    Looks.Hide();
    Looks.SwitchCostumeTo(costume1);
    Forever
    {
        If Sensing.TouchingObject(_mouse_)
        {
            Looks.SwitchCostumeTo(costume2);
        }
        Else
        {
            Looks.SwitchCostumeTo(costume1);
        }
    }
}


//----- Sprite clicked events -----------------------------------------------------------------------------------------

WhenThisSpriteClicked()
{
    If (fntme == 1)
    {
        Event.Broadcast("enter funtime auditorium");
        Stop(this script);
    }
    Else
    {
        If (prm == 1)
        {
            Event.Broadcast("primary control module");
            Stop(this script);
        }
    }
}


//----- Broadcast received events -------------------------------------------------------------------------------------

WhenBroadcastReceived(night 1)
{
    Looks.Hide();
}

WhenBroadcastReceived(night 1)
{
    Repeat (10)
    {
        Control.Wait(0.1);
        Looks.ChangeEffectBy(GHOST, 10);
    }
}

WhenBroadcastReceived(primary control module)
{
    Looks.Hide();
}

WhenBroadcastReceived(show)
{
    Looks.Show();
    Looks.SetEffectTo(GHOST, 40);
}
