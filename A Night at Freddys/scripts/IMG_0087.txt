//----- Green flag events ---------------------------------------------------------------------------------------------

WhenGreenFlagClicked()
{
    aggression factor = 0;
    Looks.Hide();
}

WhenGreenFlagClicked()
{
    Forever
    {
        If ((Not ((aggression factor == 1))) Or (Not ((aggression factor == 6))))
        {
            Looks.Hide();
            Stop(this script);
        }
    }
}


//----- Key pressed events --------------------------------------------------------------------------------------------

WhenKeyPressed(space)
{
    Wait Until (funtime auditorium == 1);
    Control.Wait(0.5);
    Wait Until ((aggression factor == 1) Or (aggression factor == 6));
    Looks.Show();
    Repeat (43)
    {
        Looks.NextCostume();
    }
    Looks.Hide();
    Stop(this script);
}

WhenKeyPressed(space)
{
    aggression factor += 1;
}


//----- Broadcast received events -------------------------------------------------------------------------------------

WhenBroadcastReceived(scooping room)
{
    Looks.Hide();
}
