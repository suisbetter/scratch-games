//----- Green flag events ---------------------------------------------------------------------------------------------

WhenGreenFlagClicked()
{
    Looks.SetSizeTo(190);
    Motion.GoToXY(0, 0);
    Looks.Hide();
}


//----- Broadcast received events -------------------------------------------------------------------------------------

WhenBroadcastReceived(door opening)
{
    Looks.SetSizeTo(190);
    Looks.Show();
    Control.Wait(0.5);
    Sound.Play(Five Nights At Freddy's_ Sister Location Door Sound);
    Repeat (12)
    {
        Looks.NextCostume();
    }
}

WhenBroadcastReceived(primary control module)
{
    Looks.Hide();
}
