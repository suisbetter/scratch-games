//----- Green flag events ---------------------------------------------------------------------------------------------

WhenGreenFlagClicked()
{
    circus control 2 = 0;
    Looks.SetEffectTo(GHOST, 100);
    Looks.SetSizeTo(130);
    Motion.GoToXY(0, 0);
    Looks.Hide();
}

WhenGreenFlagClicked()
{
    Forever
    {
        Motion.SetX(((Motion.XPosition() - ((Sensing.MouseX() + Motion.XPosition()) / 40)) / 1.18));
    }
}

WhenGreenFlagClicked()
{
    Forever
    {
        Motion.SetY(((Motion.YPosition() - ((Sensing.MouseY() + Motion.YPosition()) / 40)) / 1.18));
    }
}


//----- Key pressed events --------------------------------------------------------------------------------------------

WhenKeyPressed(b)
{
    Event.Broadcast("go back");
    Repeat (10)
    {
        Control.Wait(0.01);
        Looks.ChangeEffectBy(GHOST, 10);
    }
}


//----- Broadcast received events -------------------------------------------------------------------------------------

WhenBroadcastReceived(hide)
{
    Looks.Hide();
}

WhenBroadcastReceived(in circus control)
{
    Sound.Play(Door-Me_58G_HD2-29632);
    Looks.Show();
    Looks.SetSizeTo(140);
    Repeat (10)
    {
        Control.Wait(0.01);
        Looks.ChangeEffectBy(GHOST, -10);
    }
    Sound.PlayUntilDone(recording1);
    Control.Wait(2);
    Sound.PlayUntilDone(Circus_Baby_Dialogue_-_Night_2_-_Part_1);
    Control.Wait(2);
    Sound.PlayUntilDone(recording2);
    Control.Wait(2);
    Sound.PlayUntilDone(Circus_Baby_Dialogue_-_Night_3_-_Part_1);
    Control.Wait(2);
    Sound.PlayUntilDone(recording3);
}
