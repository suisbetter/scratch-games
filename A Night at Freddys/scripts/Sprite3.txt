//----- Variables -----------------------------------------------------------------------------------------------------

1 = 0
2 = 0
3 = 1086
4 = 1
5 = 720
6 = 760
Length = 1092



//----- Green flag events ---------------------------------------------------------------------------------------------

WhenGreenFlagClicked()
{
    Pen.Clear();
}

WhenGreenFlagClicked()
{
    go = 0;
}


//----- Broadcast received events -------------------------------------------------------------------------------------

WhenBroadcastReceived(night 1)
{
    Control.Wait(23);
    go += 1;
    Pen.Clear();
}

WhenBroadcastReceived(night 1)
{
    Control.Wait(17);
    Pen.Clear();
    Looks.Hide();
    Pen.Clear();
    Pen.SetPenSizeTo(10);
    1 = 0;
    Length = 0;
    5 = 0;
    6 = 0;
    Sensing.ResetTimer();
    Repeat Until (go == 1)
    {
        Pen.Clear();
        2 = 1;
        3 = Length;
        Call Draw;
        1 = 2;
        Length = 3;
        Length += 6;
        If ((sin(3) < sin(Length)) Or (sin(Length) == sin(3)))
        {
            4 = 1;
        }
        Else
        {
            4 = 0;
        }
    }
}


//----- Custom blocks -------------------------------------------------------------------------------------------------

Define Draw (warp=true)
{
    Pen.SetPenColorToColor(#e2e2e2);
    Motion.GoToXY((sin(((1 + 5) + Length)) * 30), (cos(((1 + 5) + Length)) * 30));
    If (4 == 1)
    {
        Pen.Down();
    }
    Repeat (Operator.Round(((sin(Length) * 120) + 160)))
    {
        Motion.GoToXY((sin(((1 + 5) + Length)) * 30), (cos(((1 + 5) + Length)) * 30));
        1 += 1;
    }
    Pen.Clear();
    If (abs(((Length % 360) - 90)) < 4)
    {
        6 = (1 + 5);
    }
    1 = 0;
    Motion.GoToXY((sin(((1 + 6) + Length)) * 30), (cos(((1 + 6) + Length)) * 30));
    Pen.Down();
    If (4 == 0)
    {
        Repeat (Operator.Round(((sin(Length) * 120) + 160)))
        {
            Motion.GoToXY((sin(((1 + 6) + Length)) * 30), (cos(((1 + 6) + Length)) * 30));
            1 += -1;
        }
    }
    Pen.Clear();
    If (abs(((Length % 360) - 270)) < 4)
    {
        5 = (1 + 6);
    }
}
