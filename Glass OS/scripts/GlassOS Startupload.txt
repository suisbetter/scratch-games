//----- Green flag events ---------------------------------------------------------------------------------------------

WhenGreenFlagClicked()
{
    Verification = 0;
    Test = 0;
    Game = 0;
    When Pin 0 Pressed = 0;
    When Pin 1 Pressed = 0;
    Open Test App = 0;
    OS Opening = 0;
    AM = 0;
    AM or PM = 0;
    Hide = 0;
    Event.Broadcast("Hide");
    Looks.SwitchBackdropTo(backdrop1);
    Sensing.Ask("AM or PM?");
    Looks.SetEffectTo(COLOR, 0);
    Motion.GoToXY(0, 0);
    Looks.Hide();
    Control.Wait(1);
    Looks.Show();
    Repeat (100)
    {
        Looks.ChangeEffectBy(COLOR, -3);
    }
    Looks.Hide();
    Event.Broadcast("OS Opening");
}

WhenGreenFlagClicked()
{
    Forever
    {
        AM or PM = Sensing.Answer();
    }
}
