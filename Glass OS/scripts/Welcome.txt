//----- Green flag events ---------------------------------------------------------------------------------------------

WhenGreenFlagClicked()
{
    Motion.GoToXY(0, 0);
    Looks.SetEffectTo(GHOST, 100);
    Looks.Hide();
}


//----- Broadcast received events -------------------------------------------------------------------------------------

WhenBroadcastReceived(Hide)
{
    Looks.Hide();
}

WhenBroadcastReceived(Load1Done)
{
    Looks.SwitchCostumeTo(Welcome);
    Control.Wait(2.3);
    Looks.Show();
    Repeat (100)
    {
        Looks.ChangeEffectBy(GHOST, -5);
    }
    Control.Wait(1);
    Repeat (27)
    {
        Looks.NextCostume();
    }
}

WhenBroadcastReceived(Loading Guest mode)
{
    Event.Broadcast("Hide 2 Sprites");
    Repeat (27)
    {
        Looks.SwitchCostumeTo((Costume.GetNumberName(number) - 1));
    }
    Repeat (100)
    {
        Looks.ChangeEffectBy(GHOST, 5);
    }
    Event.Broadcast("LGM Complete");
}
