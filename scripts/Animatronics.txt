//----- Custom blocks -------------------------------------------------------------------------------------------------

Define rng man(string goto) (warp=false)
{
    Looks.SwitchCostumeTo(man);
    goto man = goto;
    previous cam man = goto man;
    List.Add(manlist, Operator.Join("CAM", (Operator.LetterOf(goto man, 4) + -1)));
    List.Add(manlist, Operator.Join("CAM", (Operator.LetterOf(goto man, 4) + -2)));
    goto man = manlist[Operator.Random(List.IndexOf(manlist, manlist[2]) , List.IndexOf(manlist, manlist[1]))];
    List.ReplaceItem(camlist, Operator.LetterOf(goto man, 4), Operator.Join(Costume.GetNumberName(name), Operator.Join("(", Operator.Join(goto man, ")"))));
    List.ReplaceItem(camlist, Operator.LetterOf(previous cam man, 4), goto);
    wait = 0;
    Control.Wait(1);
    List.DeleteAll(manlist);
    Looks.SwitchCostumeTo(1);
}

Define rng women(string goto) (warp=false)
{
    Looks.SwitchCostumeTo(women1);
    goto women = goto;
    previous cam woman = goto women;
    List.Add(womanlist, Operator.Join("CAM", (Operator.LetterOf(goto women, 4) + -1)));
    List.Add(womanlist, Operator.Join("CAM", (Operator.LetterOf(goto women, 4) + -2)));
    goto women = womanlist[Operator.Random(List.IndexOf(womanlist, womanlist[2]) , List.IndexOf(womanlist, womanlist[1]))];
    currentwomen = goto women;
    List.ReplaceItem(camlist, Operator.LetterOf(goto women, 4), Operator.Join(Costume.GetNumberName(name), Operator.Join("(", Operator.Join(goto women, ")"))));
    wait = 0;
    Control.Wait(1);
    List.DeleteAll(womanlist);
    Looks.SwitchCostumeTo(1);
}
